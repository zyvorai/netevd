6use anyhow::{Context, Result};
7use tracing::{debug, info};
8use zbus::Connection;

10const RESOLVED_SERVICE: &str = "org.freedesktop.resolve1";
11const RESOLVED_PATH: &str = "/org/freedesktop/resolve1";
12const RESOLVED_INTERFACE: &str = "org.freedesktop.resolve1.Manager";
13
14/// Set DNS servers for a specific link via systemd-resolved
15pub async fn set_link_dns(ifindex: u32, dns_servers: Vec<String>) -> Result<()> {
16    if dns_servers.is_empty() {
17        debug!("No DNS servers to set for interface {}", ifindex);
18        return Ok(());
19    }
20
21    info!(
22        "Setting DNS servers for interface {}: {:?}",
23        ifindex, dns_servers
24    );
25
26    let connection = get_system_bus().await?;
27
28    let dns_array: Vec<(i32, Vec<u8>)> = dns_servers
29        .iter()
30        .filter_map(|server| {
31            if let Ok(addr) = server.parse::<std::net::IpAddr>() {
32                let ifindex_i32 = i32::try_from(ifindex).ok()?;
33                let _ = ifindex_i32; // validated
34                match addr {
35                    std::net::IpAddr::V4(ipv4) => Some((2, ipv4.octets().to_vec())),
36                    std::net::IpAddr::V6(ipv6) => Some((10, ipv6.octets().to_vec())),
37                }
38            } else {
39                None
40            }
41        })
42        .collect();
43
44    if dns_array.is_empty() {
45        debug!("No valid DNS servers parsed");
46        return Ok(());
47    }
48
49    let ifindex_i32 = i32::try_from(ifindex).context("ifindex out of i32 range")?;
50
51    let proxy = zbus::Proxy::new(
52        &connection,
53        RESOLVED_SERVICE,
54        RESOLVED_PATH,
55        RESOLVED_INTERFACE,
56    )
57    .await
58    .context("Failed to create resolved proxy")?;
59
60    proxy
61        .call_method("SetLinkDNS", &(ifindex_i32, dns_array))
62        .await
63        .context("Failed to call SetLinkDNS")?;
64
65    info!("Successfully set DNS servers for interface {}", ifindex);
66    Ok(())
67}

69/// Set DNS domains for a specific link via systemd-resolved
70pub async fn set_link_domains(ifindex: u32, domains: Vec<String>) -> Result<()> {
71    if domains.is_empty() {
72        debug!("No domains to set for interface {}", ifindex);
73        return Ok(());
74    }
75
76    info!(
77        "Setting DNS domains for interface {}: {:?}",
78        ifindex, domains
79    );
80
81    let connection = get_system_bus().await?;
82    let ifindex_i32 = i32::try_from(ifindex).context("ifindex out of i32 range")?;
83
84    let domain_array: Vec<(String, bool)> = domains.iter().map(|d| (d.clone(), false)).collect();
85
86    let proxy = zbus::Proxy::new(
87        &connection,
88        RESOLVED_SERVICE,
89        RESOLVED_PATH,
90        RESOLVED_INTERFACE,
91    )
92    .await
93    .context("Failed to create resolved proxy")?;
94
95    proxy
96        .call_method("SetLinkDomains", &(ifindex_i32, domain_array))
97        .await
98        .context("Failed to call SetLinkDomains")?;
99
100    info!("Successfully set DNS domains for interface {}", ifindex);
101    Ok(())
102}

104/// Get or create a cached system bus connection
105async fn get_system_bus() -> Result<Connection> {
106    use std::sync::OnceLock;
107    use tokio::sync::Mutex;
108
109    static BUS: OnceLock<Mutex<Option<Connection>>> = OnceLock::new();
110    let lock = BUS.get_or_init(|| Mutex::new(None));
111    let mut guard = lock.lock().await;
112
113    if let Some(ref conn) = *guard {
114        // Check if connection is still alive by trying a basic operation
115        if conn.is_bus() {
116            return Ok(conn.clone());
117        }
118    }
119
120    let conn = Connection::system()
121        .await
122        .context("Failed to connect to system bus")?;
123    *guard = Some(conn.clone());
124    Ok(conn)
125}
