4//! Coalesce bursty netlink events per (link, event) key.
5
6use std::collections::HashMap;
7use std::time::{Duration, Instant};
8
9use super::event::HookEventV1;
10
11#[derive(Debug, Clone)]
12struct Pending {
13    event: HookEventV1,
14    first_seen: Instant,
15    last_seen: Instant,
16    count: u32,
17}
18
19#[derive(Debug)]
20pub struct HookDebouncer {
21    window: Duration,
22    pending: HashMap<String, Pending>,
23    coalesced_total: u64,
24}
25
26impl HookDebouncer {
27    pub fn new(window: Duration) -> Self {
28        Self {
29            window,
30            pending: HashMap::new(),
31            coalesced_total: 0,
32        }
33    }
34
35    pub fn from_millis(ms: u64) -> Self {
36        Self::new(Duration::from_millis(ms))
37    }
38
39    fn key(event: &HookEventV1) -> String {
40        format!("{}:{}", event.link, event.event)
41    }
42
43    /// Push an event. Returns the previous pending event if this push
44    /// starts a *new* window after the old one already expired (caller
45    /// should dispatch that previous one immediately).
46    pub fn push(&mut self, event: HookEventV1) -> Option<HookEventV1> {
47        let now = Instant::now();
48        let key = Self::key(&event);
49        match self.pending.get_mut(&key) {
50            Some(p) if now.duration_since(p.first_seen) <= self.window => {
51                self.coalesced_total += 1;
52                p.count += 1;
53                p.last_seen = now;
54                p.event = event;
55                None
56            }
57            Some(_) => {
58                let old = self.pending.remove(&key).unwrap();
59                self.pending.insert(
60                    key,
61                    Pending {
62                        event,
63                        first_seen: now,
64                        last_seen: now,
65                        count: 1,
66                    },
67                );
68                Some(old.event)
69            }
70            None => {
71                self.pending.insert(
72                    key,
73                    Pending {
74                        event,
75                        first_seen: now,
76                        last_seen: now,
77                        count: 1,
78                    },
79                );
80                None
81            }
82        }
83    }
84
85    /// Drain events whose window has elapsed.
86    pub fn take_ready(&mut self) -> Vec<HookEventV1> {
87        let now = Instant::now();
88        let window = self.window;
89        let mut ready_keys = Vec::new();
90        for (k, p) in &self.pending {
91            if now.duration_since(p.first_seen) >= window {
92                ready_keys.push(k.clone());
93            }
94        }
95        let mut out = Vec::with_capacity(ready_keys.len());
96        for k in ready_keys {
97            if let Some(p) = self.pending.remove(&k) {
98                out.push(p.event);
99            }
100        }
101        out
102    }
103
104    /// Drain everything (shutdown / tests).
105    pub fn take_all(&mut self) -> Vec<HookEventV1> {
106        self.pending.drain().map(|(_, p)| p.event).collect()
107    }
108
109    pub fn coalesced_total(&self) -> u64 {
110        self.coalesced_total
111    }
112
113    pub fn pending_len(&self) -> usize {
114        self.pending.len()
115    }
116}
117
118/// Pure helper used by tests: N events in one window collapse to 1.
119pub fn coalesce_count(n: u32) -> u32 {
120    if n == 0 {
121        0
122    } else {
123        1
124    }
125}

127#[cfg(test)]
128mod tests {
129    use super::*;
130    use crate::hooks::event::HookEventV1;
131    use std::thread;
132
133    fn ev(link: &str, event: &str) -> HookEventV1 {
134        HookEventV1::new(event, link, 2, "systemd-networkd")
135    }
136
137    #[test]
138    fn burst_coalesces_same_key() {
139        let mut d = HookDebouncer::from_millis(80);
140        for _ in 0..20 {
141            let flushed = d.push(ev("eth0", "routable"));
142            assert!(flushed.is_none());
143        }
144        assert_eq!(d.pending_len(), 1);
145        assert_eq!(d.coalesced_total(), 19);
146        assert_eq!(coalesce_count(20), 1);
147    }
148
149    #[test]
150    fn different_keys_stay_separate() {
151        let mut d = HookDebouncer::from_millis(80);
152        d.push(ev("eth0", "routable"));
153        d.push(ev("eth0", "routes"));
154        d.push(ev("eth1", "routable"));
155        assert_eq!(d.pending_len(), 3);
156        assert_eq!(d.coalesced_total(), 0);
157    }
158
159    #[test]
160    fn take_ready_after_window() {
161        let mut d = HookDebouncer::from_millis(20);
162        d.push(ev("eth0", "address-added"));
163        thread::sleep(Duration::from_millis(35));
164        let ready = d.take_ready();
165        assert_eq!(ready.len(), 1);
166        assert_eq!(ready[0].event, "address-added");
167        assert_eq!(d.pending_len(), 0);
168    }
169
170    #[test]
171    fn last_payload_wins() {
172        let mut d = HookDebouncer::from_millis(80);
173        d.push(ev("eth0", "mtu").with_mtu(1500));
174        d.push(ev("eth0", "mtu").with_mtu(9000));
175        let all = d.take_all();
176        assert_eq!(all.len(), 1);
177        assert_eq!(all[0].mtu, Some(9000));
178    }
179}
