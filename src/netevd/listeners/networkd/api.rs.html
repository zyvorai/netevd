11const SYSTEMD_NETIF_LINKS: &str = "/run/systemd/netif/links";
12const SYSTEMD_NETIF_STATE: &str = "/run/systemd/netif/state";

14/// Link state information from systemd-networkd

17    pub admin_state: String,
18    pub oper_state: String,
19    pub carrier_state: String,

41/// Parse a flat KEY=VALUE file into a HashMap
42fn parse_key_value_file(path: &std::path::Path) -> Result<HashMap<String, String>> {
43    let contents = std::fs::read_to_string(path)
44        .map_err(|e| anyhow::anyhow!("Failed to read file {:?}: {}", path, e))?;
45
46    let mut kv = HashMap::new();
47    for line in contents.lines() {
48        let line = line.trim();
49        if line.is_empty() || line.starts_with('#') {
50            continue;
51        }
52        if let Some((key, value)) = line.split_once('=') {
53            kv.insert(key.to_string(), value.to_string());
54        }
55    }
56    Ok(kv)
57}

59/// Parse link state file from systemd-networkd
60pub fn parse_link_state_file(ifindex: u32) -> Result<LinkState> {
61    let path = PathBuf::from(SYSTEMD_NETIF_LINKS).join(ifindex.to_string());
62
63    if !path.exists() {
64        debug!("Link state file does not exist: {:?}", path);
65        return Ok(LinkState::default());
66    }
67
68    let kv = parse_key_value_file(&path)?;
69
70    let mut dns = Vec::new();
71    if let Some(dns_str) = kv.get("DNS") {
72        dns = dns_str.split_whitespace().map(|s| s.to_string()).collect();
73    }
74
75    let mut domains = Vec::new();
76    if let Some(domains_str) = kv.get("DOMAINS") {
77        domains = domains_str
78            .split_whitespace()
79            .map(|s| s.to_string())
80            .collect();
81    }
82
83    let state = LinkState {
84        admin_state: kv.get("ADMIN_STATE").cloned().unwrap_or_default(),
85        oper_state: kv.get("OPER_STATE").cloned().unwrap_or_default(),
86        carrier_state: kv.get("CARRIER_STATE").cloned().unwrap_or_default(),
87        address_state: kv.get("ADDRESS_STATE").cloned().unwrap_or_default(),
88        ipv4_address_state: kv.get("IPV4_ADDRESS_STATE").cloned().unwrap_or_default(),
89        ipv6_address_state: kv.get("IPV6_ADDRESS_STATE").cloned().unwrap_or_default(),
90        online_state: kv.get("ONLINE_STATE").cloned().unwrap_or_default(),
91        dns,
92        domains,
93        gateway: kv.get("GATEWAY").cloned().filter(|s| !s.is_empty()),
94        gateway6: kv.get("GATEWAY6").cloned().filter(|s| !s.is_empty()),
95    };
96
97    debug!("Parsed link state for ifindex {}: {:?}", ifindex, state);
98    Ok(state)
99}

101/// Parse manager state file from systemd-networkd

104

111

114        carrier_state: kv.get("CARRIER_STATE").cloned().unwrap_or_default(),
115        address_state: kv.get("ADDRESS_STATE").cloned().unwrap_or_default(),
116        ipv4_address_state: kv.get("IPV4_ADDRESS_STATE").cloned().unwrap_or_default(),
