4use crate::api::handlers::AppState;
5use crate::api::routes::create_api_routes;
6use crate::network::NetworkState;
7use anyhow::Result;
8use axum::http::{HeaderValue, Method};
9use std::net::SocketAddr;
10use std::sync::Arc;
11use tokio::sync::RwLock;
12use tower_http::cors::{AllowOrigin, CorsLayer};
13use tower_http::trace::TraceLayer;
14
15pub struct ApiServer {
16    addr: SocketAddr,
17    state: AppState,
18}
19
20impl ApiServer {
21    pub fn new(port: u16, state: Arc<RwLock<NetworkState>>) -> Self {
22        let addr = SocketAddr::from(([127, 0, 0, 1], port));
23        Self { addr, state }
24    }
25
26    pub async fn run(self) -> Result<()> {
27        let cors = CorsLayer::new()
28            .allow_origin(AllowOrigin::predicate(|origin: &HeaderValue, _| {
29                origin.to_str().is_ok_and(|s| {
30                    s == "http://127.0.0.1"
31                        || s.starts_with("http://127.0.0.1:")
32                        || s == "http://localhost"
33                        || s.starts_with("http://localhost:")
34                        || s == "http://[::1]"
35                        || s.starts_with("http://[::1]:")
36                })
37            }))
38            .allow_methods([Method::GET, Method::POST])
39            .allow_headers([axum::http::header::CONTENT_TYPE]);
40
41        let app = create_api_routes(self.state)
42            .layer(cors)
43            .layer(TraceLayer::new_for_http());
44
45        tracing::info!("Starting API server on {}", self.addr);
46
47        let listener = tokio::net::TcpListener::bind(self.addr).await?;
48        axum::serve(listener, app).await?;
49
50        Ok(())
51    }
52}
