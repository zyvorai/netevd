4use chrono::{DateTime, Utc};
5use serde::{Deserialize, Serialize};
6use std::net::IpAddr;

8#[derive(Debug, Serialize, Deserialize, Clone)]
9pub struct ApiResponse<T> {
10    pub success: bool,
11    pub data: Option<T>,
12    pub error: Option<String>,
13    pub timestamp: DateTime<Utc>,
14}
15
16impl<T> ApiResponse<T> {
17    pub fn success(data: T) -> Self {
18        Self {
19            success: true,
20            data: Some(data),
21            error: None,
22            timestamp: Utc::now(),
23        }
24    }
25
26    pub fn error(message: String) -> Self {
27        Self {
28            success: false,
29            data: None,
30            error: Some(message),
31            timestamp: Utc::now(),
32        }
33    }
34}
